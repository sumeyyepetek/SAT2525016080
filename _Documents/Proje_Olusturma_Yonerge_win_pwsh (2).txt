1) Gereksinimleri kontrol etme

Terminal: Windows VS Code PowerShell. 
Dosya içerikleri terminalden yazılmaz; 
VS Code editöründe kopyalanıp yapıştırılır.

.NET 8 SDK (veya net8.0 şablonunu destekleyen bir SDK), 
SQL Server ve sa hesabı gerekir. 
Örnek sunucu localhost,1445

dotnet --list-sdks
dotnet tool install --global dotnet-ef --version 8.0.24
dotnet ef --version

dotnet-ef zaten kuruluysa: 
dotnet tool update --global dotnet-ef --version 8.0.24

Ana çalışma klasöründe kalarak projeyi doğrudan şablondan oluşturun:

dotnet new blazor -n BlazorApp -o BlazorApp -f net8.0 --interactivity Server --all-interactive --auth Individual --use-local-db --no-restore

Set-Location 'BlazorApp'

Bundan sonraki tüm komutları BlazorApp klasöründe çalıştırın. 

Şablonun ürettiği Components/Account, App.razor, MainLayout.razor ve diğer örnek bileşenler aynen kalır.

using Entities;
using Microsoft.EntityFrameworkCore;


namespace DbContexts;

public class XDbContext(DbContextOptions<XDbContext> options) : DbContext(options)
{

    // DbSet özellikleri
    public DbSet<Department> Departments { get; set; }
    public DbSet<Student> Students { get; set; }


    // Model oluşturma metodu
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        // Temel model ayarları
        base.OnModelCreating(modelBuilder);

        // Tablolar ve şema
        modelBuilder.Entity<Department>().ToTable("Department", "School");
        modelBuilder.Entity<Student>().ToTable("Student", "School");

        // Bölüm ilişkisi
        modelBuilder.Entity<Student>()
            .HasOne(student => student.Department)
            .WithMany(department => department.Students)
            .HasForeignKey(student => student.DepartmentId)
            .OnDelete(DeleteBehavior.Cascade)
            .HasConstraintName("FK_Student_Department");

        // Öğrenci sütun ayarları
        modelBuilder.Entity<Student>()
            .Property(student => student.FirstName)
            .HasMaxLength(50);

        modelBuilder.Entity<Student>()
            .Property(student => student.LastName)
            .HasMaxLength(50);

        modelBuilder.Entity<Student>()
            .Property(student => student.Gender)
            .HasDefaultValue(0);

        // Öğrenci ad indeksi
        modelBuilder.Entity<Student>()
            .HasIndex(student => new { student.FirstName, student.LastName })
            .HasDatabaseName("IX_Student_FirstNameLastName");
    }
}

using System.Reflection;
using Attributes;

namespace Extensions;

public static class EnumExtensions
{

    // Metotlar
    public static T? GetAttr<T>(this Enum item)
        where T : Attribute => item.GetType().GetField(item.ToString())?.GetCustomAttribute<T>();
    public static T? GetAttribute<T>(this Enum field)
        where T : Attribute => field.GetType().GetField(field.ToString())?.GetCustomAttribute<T>();
    public static string Color(this Enum field) => field.GetAttribute<ColorAttribute>()?.Value ?? "";
    public static string DisplayText(this Enum field) => field.GetAttribute<DisplayTextAttribute>()?.Value ?? field.ToString();
    public static string Symbol(this Enum field) => field.GetAttribute<SymbolAttribute>()?.Value ?? field.ToString();
    public static string SymbolDisplayText(this Enum field) => $"{field.Symbol()} {field.DisplayText()}";

    public static int ToInt<TEnum>(this TEnum value)
        where TEnum : struct, Enum => Convert.ToInt32(value);
    public static TEnum ToEnum<TEnum>(this int value)
        where TEnum : struct, Enum => (TEnum)Enum.ToObject(typeof(TEnum), value);
    public static TEnum ToEnum<TEnum>(this byte value)
        where TEnum : struct, Enum => ((int)value).ToEnum<TEnum>();
}

@inject IDbContextFactory<XDbContext> dbContextFactory

@code {

    // Alanlar
    Operation operation = Operation.List;
    IEnumerable<Department> departments = [];
    Department currentDepartment = new Department();

    // Geçersiz kılınan metotlar
    protected override async Task OnInitializedAsync()
    {
        await GetItemsAsync();
    }

    // Metotlar
    private async Task GetItemsAsync()
    {
        using var db = await dbContextFactory.CreateDbContextAsync();
        departments = await db.Departments.ToListAsync();
    }

    private void Clear()
    {
        currentDepartment = new Department();
        operation = Operation.List;
    }

    private async Task OnSaveAsync()
    {
        using var db = await dbContextFactory.CreateDbContextAsync();
        if (operation == Operation.Insert && currentDepartment.Id == 0)
        {
            currentDepartment.CreatedBy = "abc";
            currentDepartment.CreatedDateTime = DateTime.Now;
            await db.Departments.AddAsync(currentDepartment);
        }

        if (operation == Operation.Delete)
        {
            db.Departments.Remove(currentDepartment);
        }

        if (operation == Operation.Update)
        {
            db.Departments.Update(currentDepartment);
        }

        await db.SaveChangesAsync();
        await GetItemsAsync();
        Clear();
    }

    private void OnNew()
    {
        operation = Operation.Insert;
        currentDepartment = new Department();
    }

    private void OnUpdate(Department department)
    {
        operation = Operation.Update;
        currentDepartment = department;
    }

    private void OnDelete(Department department)
    {
        operation = Operation.Delete;
        currentDepartment = department;
    }

    private void OnCancel()
    {
        currentDepartment = new Department();
        operation = Operation.List;
    }
}

@using DbContexts
@using Entities
@using Enums
@using Extensions
@using Microsoft.Data.SqlClient
@using Microsoft.EntityFrameworkCore

@inject IDbContextFactory<XDbContext> dbContextFactory

@code {

    // Alanlar
    // Düzenleme durumu
    private Operation _operation = Operation.List;
    private Student _currentStudent = new();

    // Veri koleksiyonları
    private IEnumerable<Student> _students = [];
    private IEnumerable<Department> _departments = [];

    // Sıralama seçimi
    private string _selectedSortProperty = nameof(Student.Id);
    private SortOrder _sortOrder = SortOrder.Ascending;

    // Sayfalama
    private int _pageSize = 5;
    private int _pageNumber = 1;
    private int _totalPageCount = 1;

    // Durum
    private string _message = "Veriler yükleniyor...";

    // İlk etkileşimli çizimden sonra öğrenci listesini yükleme
    protected override async Task OnAfterRenderAsync(bool firstRender)
    {
        if (!firstRender)
            return;

        await GetItemsAsync();
        _message = "";
        StateHasChanged();
    }

    // Metotlar
    private async Task GetDepartmentsAsync()
    {
        using var db = await dbContextFactory.CreateDbContextAsync();
        _departments = await db.Departments.ToListAsync();
    }

    private async Task GetItemsAsync(int pn)
    {
        _pageNumber = pn;
        await GetItemsAsync();
    }

    private async Task GetItemsAsync()
    {
        using var db = await dbContextFactory.CreateDbContextAsync();

        // Öğrenci sorgusu
        var items = db.Students.AsQueryable();

        // Özellik ve yöne göre sıralama
        items = SortItems(items);

        // Sayfalama toplamları
        var totalItemCount = await items.CountAsync();
        _totalPageCount = Math.Max(1, (int)Math.Ceiling(totalItemCount / (double)_pageSize));
        _pageNumber = Math.Clamp(_pageNumber, 1, _totalPageCount);

        // Sayfa seçimi
        var skip = (_pageNumber - 1) * _pageSize;
        var take = _pageSize;
        _students = await items
            .Skip(skip)
            .Take(take)
            .Include(s => s.Department)
            .ToListAsync();
    }

    private IQueryable<Student> SortItems(IQueryable<Student> items)
    => _selectedSortProperty switch
    {
        nameof(Student.Id) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.Id)
            : items.OrderByDescending(student => student.Id),

        nameof(Student.FirstName) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.FirstName).ThenBy(student => student.LastName).ThenBy(student => student.Id)
            : items.OrderByDescending(student => student.FirstName).ThenByDescending(student => student.LastName),

        nameof(Student.LastName) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.LastName).ThenBy(student => student.FirstName).ThenBy(student => student.Id)
            : items.OrderByDescending(student => student.LastName).ThenByDescending(student => student.FirstName),

        nameof(Student.Gender) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.Gender)
            : items.OrderByDescending(student => student.Gender),

        nameof(Student.DepartmentId) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.DepartmentId)
            : items.OrderByDescending(student => student.DepartmentId),

        nameof(Student.Department) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.Department!.Name)
            : items.OrderByDescending(student => student.Department!.Name),

        nameof(Student.CreatedBy) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.CreatedBy)
            : items.OrderByDescending(student => student.CreatedBy),

        nameof(Student.CreatedDateTime) => _sortOrder == SortOrder.Ascending
            ? items.OrderBy(student => student.CreatedDateTime)
            : items.OrderByDescending(student => student.CreatedDateTime),

        _ => items.OrderBy(student => student.Id),
    };

    private async Task OnSaveAsync()
    {
        using var db = await dbContextFactory.CreateDbContextAsync();

        switch (_operation)
        {
            case Operation.Insert:
                await db.Students.AddAsync(_currentStudent);
                break;
            case Operation.Delete:
                db.Students.Remove(_currentStudent);
                break;
            case Operation.Update:
                var student = await db.Students.FindAsync(_currentStudent.Id);
                if (student != null)
                    db.Entry(student).CurrentValues.SetValues(_currentStudent);
                break;
        }

        // Kaydetme sonucu
        var result = await db.SaveChangesAsync();
        _message = $"{_operation.DisplayText()} işlemi: {result} kayıt etkilendi.";
        await GetItemsAsync();
        _operation = Operation.List;
    }

    private async Task OnCancelAsync()
    {
        if (_operation == Operation.Update)
            await GetItemsAsync();

        _currentStudent = new();
        _operation = Operation.List;
    }

    private async Task OnOperationAsync(Operation selectedOperation, Student? student = null)
    {
        if (selectedOperation is Operation.Insert or Operation.Update)
            await GetDepartmentsAsync();

        _currentStudent = selectedOperation == Operation.Insert
            ? new Student()
            : student ?? throw new ArgumentNullException(nameof(student));
            
        _operation = selectedOperation;
    }

    private Task OnSortChangedAsync() => GetItemsAsync(1);
    private Task OnPageSizeChangedAsync() => GetItemsAsync(1);
    private Task OnPageNumberChangedAsync() => GetItemsAsync();

}
